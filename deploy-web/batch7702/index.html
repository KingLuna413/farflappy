// SPDX-License-Identifier: MIT
// Batch7702 - EXPERIMENT ONLY, for a dedicated burner wallet.
// This contract is designed to be used via EIP-7702 delegation on BSC mainnet
// (Pascal hard fork). It is NOT audited, NOT for a main wallet, NOT for production.
//
// How it works, in plain terms:
// - Deploy this contract from the burner wallet. OWNER is set to the deployer
//   as an immutable value, so it is baked into the bytecode, not storage.
// - When the burner EOA delegates to this contract (EIP-7702), calls to the
//   burner address execute this code IN THE BURNER'S CONTEXT.
// - sweepAll() can ONLY be triggered by the burner itself: it requires
//   msg.sender == address(this). Anyone else calling the delegated burner
//   will fail that check, because their address is not the burner.
// - sweepAll() moves every listed ERC20 balance and all remaining native BNB
//   from the burner to the `to` address given by the burner in that call.
// - After the experiment, REVOKE the delegation by delegating the burner to
//   the zero address (0x0000000000000000000000000000000000000000). Until then,
//   this code remains the burner's code.
//
// WARNING: Delegating a wallet that holds real funds to ANY contract, even
// this one, means that contract can move those funds. Use ONLY a dedicated
// burner wallet, with only dust amounts for the test, and revoke afterwards.
pragma solidity ^0.8.24;

interface IERC20 {
    function balanceOf(address account) external view returns (uint256);
    function transfer(address to, uint256 amount) external returns (bool);
}

contract Batch7702 {
    address public immutable OWNER;

    constructor() {
        OWNER = msg.sender;
    }

    /// @notice Sweep all listed tokens + all native BNB to `to`.
    /// @dev Only works as intended via EIP-7702 delegation, when the caller
    /// is the delegated EOA itself (a tx sent FROM the burner TO the burner).
    function sweepAll(address[] calldata tokens, address payable to) external {
        require(msg.sender == address(this), "only self");
        require(to != address(0), "to zero");
        for (uint256 i = 0; i < tokens.length; i++) {
            uint256 bal = IERC20(tokens[i]).balanceOf(address(this));
            if (bal > 0) {
                require(IERC20(tokens[i]).transfer(to, bal), "token transfer failed");
            }
        }
        uint256 ethBal = address(this).balance;
        if (ethBal > 0) {
            (bool ok, ) = to.call{value: ethBal}("");
            require(ok, "eth transfer failed");
        }
    }
}
